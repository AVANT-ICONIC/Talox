# Talox v6.0 — Autonomous Loop & Dynamic Skill Creation

> **For Hermes:** Use subagent-driven-development skill to implement this plan task-by-task.

**Goal:** Build an autonomous research/interaction loop that lets Talox complete multi-step browser tasks without human intervention — observing, planning, acting, learning from blockers, writing new skills on the fly, and retrying until done or escalating to human.

**Architecture:** A new `AutonomousLoop` class sits above `TaloxController`, composing the existing perception, challenge, adaptation, and skills subsystems with a new `Planner` interface (LLM-based plan generation) and `SkillWriter` (dynamic skill creation). Zero changes to `TaloxPageState` contract (frozen v1). All new code in `src/core/loop/` and `src/core/skills/`.

**Tech Stack:** TypeScript, Playwright, OpenAI-compatible API (same as ChatSession), vitest

```typescript
// src/core/loop/Planner.ts
import type { PlannerConfig, PlannerInput, TaskPlan } from "./types.js";
import type { TaloxPageState } from "../../types/index.js";

const DEFAULT_SYSTEM_PROMPT = `You are an autonomous browser agent. Given the current page state and a goal, generate a plan.

RULES:
- Return valid JSON matching the TaskPlan schema
- Be specific with selectors — use data from the page state
- If the goal is already achieved, set goalAchieved to true with empty steps
- If blocked, describe the blocker in detail
- Each step should be a single atomic action
- Prefer the simplest path to the goal
- If a skill is provided, follow its instructions
- Never attempt actions that could cause data loss`;

export interface Planner {
	plan(input: PlannerInput): Promise<TaskPlan>;
}

export class LLMPlanner implements Planner {
	private readonly config: Required<PlannerConfig>;

	constructor(config: PlannerConfig) {
		this.config = {
			apiBaseUrl: config.apiBaseUrl ?? "https://openrouter.ai/api/v1",
			apiKey: config.apiKey ?? process.env["OPENAI_API_KEY"] ?? "",
			model: config.model,
			maxTokens: config.maxTokens ?? 2048,
			systemPrompt: config.systemPrompt ?? DEFAULT_SYSTEM_PROMPT,
			contextWindowIterations: config.contextWindowIterations ?? 5,
		};
	}

	async plan(input: PlannerInput): Promise<TaskPlan> {
		const userMessage = this.buildUserMessage(input);
		const response = await this.callLLM(userMessage);
		return this.parsePlan(response);
	}

	private buildUserMessage(input: PlannerInput): string {
		const parts: string[] = [];

		parts.push(`## Goal\n${input.goal.description}`);

		if (input.state) {
			parts.push(`## Current Page State
URL: ${input.state.url}
Title: ${input.state.title}
Interactive Elements: ${JSON.stringify(input.state.interactiveElements?.slice(0, 30) ?? [], null, 2)}
Console Errors: ${JSON.stringify(input.state.console?.errors ?? [])}
Network Errors: ${JSON.stringify(input.state.network?.failedRequests ?? [])}
Bugs: ${JSON.stringify(input.state.bugs ?? [])}`);
		}

		if (input.challengeState?.detected) {
			parts.push(`## Challenge Detected
Type: ${input.challengeState.primary?.type}
Confidence: ${input.challengeState.primary?.confidence}
Evidence: ${JSON.stringify(input.challengeState.primary?.evidence ?? [])}`);
		}

		if (input.skillsContext) {
			parts.push(`## Domain Skills\n${input.skillsContext}`);
		}

		if (input.domainHints) {
			parts.push(`## Domain Memory\n${input.domainHints}`);
		}

		if (input.recentIterations.length > 0) {
			const recent = input.recentIterations.slice(-this.config.contextWindowIterations);
			parts.push(`## Recent History
${recent.map((it) => `- Iteration ${it.iteration}: ${it.observation} → ${it.result.status}${it.result.error ? ` (${it.result.error})` : ""}`).join("\n")}`);
		}

		parts.push(`## Instructions
Analyze the current state and generate a plan. Return JSON only.`);
		return parts.join("\n\n");
	}

	private async callLLM(userMessage: string): Promise<string> {
		const response = await fetch(`${this.config.apiBaseUrl}/chat/completions`, {
			method: "POST",
			headers: {
				"Content-Type": "application/json",
				Authorization: `Bearer ${this.config.apiKey}`,
			},
			body: JSON.stringify({
				model: this.config.model,
				max_tokens: this.config.maxTokens,
				messages: [
					{ role: "system", content: this.config.systemPrompt },
					{ role: "user", content: userMessage },
				],
				response_format: { type: "json_object" },
			}),
		});

		if (!response.ok) {
			throw new Error(`LLM API error: ${response.status} ${await response.text()}`);
		}

		const data = await response.json() as {
			choices: Array<{ message: { content: string } }>;
			usage: { prompt_tokens: number; completion_tokens: number; total_tokens: number };
		};

		return data.choices[0]?.message?.content ?? "";
	}

	private parsePlan(raw: string): TaskPlan {
		try {
			const plan = JSON.parse(raw) as TaskPlan;
			// Validate required fields
			if (typeof plan.goalAchieved !== "boolean") {
				plan.goalAchieved = false;
			}
			if (!Array.isArray(plan.steps)) {
				plan.steps = [];
			}
			return plan;
		} catch {
			return {
				assessment: "Failed to parse LLM response",
				steps: [],
				goalAchieved: false,
				blocker: {
					type: "unknown",
					confidence: 1,
					description: `LLM returned invalid JSON: ${raw.slice(0, 200)}`,
					evidence: [],
					autoResolvable: false,
				},
			};
		}
	}
}
```

```typescript
// src/core/skills/SkillWriter.ts
import { mkdir, writeFile, readFile, access } from "node:fs/promises";
import { join } from "node:path";
import type { DynamicSkill } from "../loop/types.js";
import { SkillLoader } from "./SkillLoader.js";

export class SkillWriter {
	private readonly skillsDir: string;
	private readonly loader: SkillLoader;

	constructor(skillsDir: string, loader: SkillLoader) {
		this.skillsDir = skillsDir;
		this.loader = loader;
	}

	/** Create a new skill from LLM-generated content. */
	async createSkill(skill: DynamicSkill): Promise<string> {
		const dir = join(this.skillsDir, skill.domain);
		await mkdir(dir, { recursive: true });

		const filePath = join(dir, "SKILL.md");
		const content = this.formatSkillFile(skill);

		await writeFile(filePath, content, "utf-8");

		// Reload so it's available immediately
		this.loader.load(filePath);

		return filePath;
	}

	/** Validate a skill works by loading it and checking structure. */
	async validateSkill(name: string): Promise<boolean> {
		const loaded = this.loader.get(name);
		if (!loaded) return false;

		// Basic validation: has frontmatter with name + description
		if (!loaded.name || !loaded.description) return false;
		return true;
	}

	/** Check if a skill already exists for this domain. */
	async existsForDomain(hostname: string): Promise<boolean> {
		const matched = this.loader.matchDomain(hostname);
		return matched !== null;
	}

	/** Format a DynamicSkill into SKILL.md format. */
	private formatSkillFile(skill: DynamicSkill): string {
		return `---
name: ${skill.name}
description: ${skill.description}
domain: ${skill.domain}
version: "${skill.version}"
trigger: ${skill.triggerCondition}
tools:
${skill.toolUsage.map((t) => `  - ${t}`).join("\n")}
---

${skill.content}
`;
	}
}
```

```typescript
	LoopEventMap,

export class AutonomousLoop {
	private readonly controller: TaloxController;
	private readonly planner: Planner;
	private readonly skillLoader: SkillLoader;
	private readonly skillWriter: SkillWriter | null;
	private readonly perceptionStack: PerceptionStack | null;
	private readonly challengeDetector: ChallengeDetector | null;
	private readonly eventBus: EventBus<LoopEventMap> | null;
	private readonly options: AutonomousLoopOptions;

	private state: LoopState | null = null;
	private abortController: AbortController | null = null;

	constructor(controller: TaloxController, options: AutonomousLoopOptions) {
		this.controller = controller;
		this.options = options;
		this.planner = new LLMPlanner(options.planner);
		this.skillLoader = new SkillLoader(options.skillsDir ? [options.skillsDir] : undefined);
		this.skillWriter = options.skillsDir
			? new SkillWriter(options.skillsDir, this.skillLoader)
			: null;
		this.perceptionStack = null; // TODO: get from controller
		this.challengeDetector = null; // TODO: get from controller
		this.eventBus = null; // TODO: create if needed

		this.controller.on("stateChanged", () => this.onStateChanged());
		this.controller.on("humanTakeoverRequested", (e) => this.onHumanTakeoverRequested(e.reason));
	}

	/** Start the autonomous loop. */
	async run(): Promise<LoopResult> {
		this.abortController = new AbortController();
		const startTime = Date.now();

		this.state = this.createInitialState();

		this.emit("loopStarted", { goal: this.state.goal });

		try {
			// Navigate to start URL if provided
			if (this.state.goal.startUrl) {
				await this.controller.navigate(this.state.goal.startUrl);
			}

			while (this.state.status === "running") {
				// Budget checks
				if (this.exceedsBudget(startTime)) {
					this.stop("budget-exhausted");
					break;
				}

				// Run one iteration
				const iteration = await this.runIteration();
				this.state.iterations.push(iteration);
				this.state.currentIteration++;

				this.emit("loopIteration", { iteration });

				if (this.options.onProgress) {
					this.options.onProgress(iteration);
				}

				// Check if goal achieved
				if (iteration.plan.goalAchieved) {
					this.stop("goal-achieved");
					break;
				}

				// Check for unresolvable blocker
				if (iteration.plan.blocker && !iteration.plan.blocker.autoResolvable) {
					const skillCreated = await this.tryCreateSkillFromBlocker(iteration.plan.blocker);
					if (!skillCreated) {
						this.emit("humanEscalation", {
							reason: iteration.plan.blocker.description,
							state: this.state,
						});

						if (this.options.onHumanEscalation) {
							const humanResponse = await this.options.onHumanEscalation(
								iteration.plan.blocker.description,
								this.state,
							);
							if (!humanResponse) {
								this.stop("human-takeover");
								break;
							}
							// Human provided guidance — continue
						} else {
							this.stop("unresolvable-blocker");
							break;
						}
					}
				}
			}
		} catch (error) {
			this.state.stopReason = "error";
			this.state.status = "failed";
		}

		const result = this.buildResult(startTime);
		this.emit("loopStopped", { reason: this.state.stopReason ?? "error", finalState: this.state });
		return result;
	}

	/** Stop the loop gracefully. */
	stop(reason: string): void {
		if (this.state) {
			this.state.status = reason === "goal-achieved" ? "completed" : "failed";
			this.state.stopReason = reason as any;
		}
		this.abortController?.abort();
	}

	/** Get current loop state (for inspection). */
	getState(): LoopState | null {
		return this.state;
	}

	// ── Private: Iteration ──────────────────────────────────────────

	private async runIteration(): Promise<LoopIteration> {
		const iterationStart = Date.now();

		// 1. OBSERVE — get current page state
		const pageState = this.controller.getState("agent");

		// 2. Get challenge state if available
		const challengeState = this.controller.getChallengeState();

		// 3. Load domain skills
		let skillsContext = "";
		try {
			const url = new URL((pageState as any)?.url ?? "about:blank");
			skillsContext = this.skillLoader.toContextForDomain(url.hostname) ?? "";
		} catch { /* NOSONAR -- non-fatal */ }

		// 4. Get domain memory hints
		const domainHints = this.getDomainHints();

		// 5. PLAN — ask the LLM what to do
		const plan = await this.planner.plan({
			state: pageState as any,
			goal: this.state!.goal,
			recentIterations: this.state!.iterations.slice(-5),
			skillsContext,
			challengeState: challengeState ?? undefined,
			domainHints,
		});

		// 6. ACT — execute the plan steps
		let result: StepResult;
		if (plan.goalAchieved) {
			result = {
				status: "success",
				state: pageState as any,
				durationMs: Date.now() - iterationStart,
			};
		} else if (plan.steps.length === 0) {
			result = {
				status: "failed",
				error: "Planner returned no steps and goal not achieved",
				durationMs: Date.now() - iterationStart,
			};
		} else {
			result = await this.executeSteps(plan.steps);
		}

		return {
			iteration: this.state!.currentIteration + 1,
			observation: plan.assessment,
			plan,
			result,
			timestamp: new Date().toISOString(),
		};
	}

	private async executeSteps(steps: PlanStep[]): Promise<StepResult> {
		const start = Date.now();
		const step = steps[0]; // Execute first step only per iteration

		if (!step) {
			return { status: "skipped", durationMs: Date.now() - start };
		}

		try {
			let state: any;
			const args = step.args;

			switch (step.tool) {
				case "navigate":
					state = await this.controller.navigate(args["url"] as string ?? args["selector"] as string);
					break;
				case "click":
					state = await this.controller.click(args["selector"] as string);
					break;
				case "type":
					state = await this.controller.type(args["selector"] as string, args["text"] as string);
					break;
				case "scrollTo":
					await this.controller.scrollTo(args["selector"] as string);
					state = this.controller.getState("agent");
					break;
				case "screenshot":
					state = await this.controller.screenshot();
					break;
				case "getState":
					state = this.controller.getState("agent");
					break;
				case "waitForSelector":
					await this.controller.waitForSelector();
					state = this.controller.getState("agent");
					break;
				case "waitForNavigation":
					await this.controller.waitForNavigation();
					state = this.controller.getState("agent");
					break;
				case "waitForTimeout":
					await this.controller.waitForTimeout(args["ms"] as number ?? 1000);
					state = this.controller.getState("agent");
					break;
				case "evaluate":
					state = await this.controller.evaluate(args["script"] as string);
					break;
				case "extractTable":
					state = await this.controller.extractTable(args["selector"] as string);
					break;
				case "findElement":
					state = await this.controller.findElement(args["text"] as string, args["type"] as any);
					break;
				default:
					return {
						status: "failed",
						error: `Unknown tool: ${step.tool}`,
						durationMs: Date.now() - start,
					};
			}

			return {
				status: "success",
				state: state as any,
				durationMs: Date.now() - start,
			};
		} catch (error: unknown) {
			const msg = error instanceof Error ? error.message : String(error);
			return {
				status: "failed",
				error: msg,
				durationMs: Date.now() - start,
			};
		}
	}

	// ── Private: Skill Creation ─────────────────────────────────────

	private async tryCreateSkillFromBlocker(blocker: BlockerClassification): Promise<boolean> {
		if (!this.skillWriter) return false;

		// For now, only create skills for patterns we can describe
		if (blocker.type === "unknown" || blocker.type === "captcha") return false;

		try {
			// Generate a basic skill from the blocker info
			// In a full implementation, the LLM would generate the skill content
			return false; // TODO: implement LLM-generated skill content
		} catch { /* NOSONAR -- non-fatal */ }
		return false;
	}

	// ── Private: Utilities ──────────────────────────────────────────

	private createInitialState(): LoopState {
		return {
			goal: this.options.goal,
			currentIteration: 0,
			iterations: [],
			createdSkills: [],
			totalTokenUsage: {
				promptTokens: 0,
				completionTokens: 0,
				totalTokens: 0,
				estimatedCostUsd: 0,
			},
			startedAt: new Date().toISOString(),
			status: "running",
		};
	}

	private exceedsBudget(startTime: number): boolean {
		if (!this.state) return false;

		// Check iterations
		if (this.state.currentIteration >= this.state.goal.maxIterations) {
			this.state.stopReason = "max-iterations";
			return true;
		}

		// Check cost
		if (this.state.goal.maxCostUsd && this.state.totalTokenUsage.estimatedCostUsd >= this.state.goal.maxCostUsd) {
			this.state.stopReason = "max-cost";
			return true;
		}

		// Check duration
		if (this.state.goal.maxDurationSeconds) {
			const elapsed = (Date.now() - startTime) / 1000;
			if (elapsed >= this.state.goal.maxDurationSeconds) {
				this.state.stopReason = "max-duration";
				return true;
			}
		}

		return false;
	}

	private getDomainHints(): string {
		try {
			const adaptationEngine = (this.controller as any)._adaptationEngine;
			if (adaptationEngine?.domainMemory) {
				const url = new URL((this.controller.getState() as any)?.url ?? "about:blank");
				const best = adaptationEngine.domainMemory.getBestStrategy(url.hostname);
				if (best) {
					return `Previous successful strategy for this domain: ${best}`;
				}
			}
		} catch { /* NOSONAR -- non-fatal */ }
		return "";
	}

	private buildResult(startTime: number): LoopResult {
		return {
			status: this.state?.status === "completed" ? "completed" : "failed",
			goal: this.state!.goal,
			totalIterations: this.state!.currentIteration,
			totalDurationMs: Date.now() - startTime,
			totalCostUsd: this.state!.totalTokenUsage.estimatedCostUsd,
			createdSkills: this.state!.createdSkills,
			finalState: this.controller.getState("agent") as any,
			stopReason: this.state!.stopReason ?? "error",
		};
	}

	private onStateChanged(): void {
		// Trigger adaptation evaluation after state changes
	}

	private onHumanTakeoverRequested(reason: string): void {
		if (this.state) {
			this.state.status = "human-takeover";
			this.state.stopReason = "human-takeover";
		}
	}

	private emit(event: string, data: any): void {
		// Event emission — will use EventBus when wired up
	}
}
```

```typescript
/** Attempt to auto-resolve the current challenge. */
async resolveChallenge(): Promise<import("../ChallengeResolver.js").ChallengeOutcome> {
	const challengeState = this.getChallengeState();
	if (!challengeState?.primary) {
		return { resolved: true, requiresHuman: false, attempts: 0, finalStrategy: "skipped" };
	}
	const page = this._session.getPage();

	return this._challengeResolver.resolve(challengeState.primary, page);
```

```typescript
private async tryCreateSkillFromBlocker(blocker: BlockerClassification): Promise<boolean> {
	if (!this.skillWriter) return false;

	// Ask the planner to generate a skill for this blocker
	const skillPrompt = `Given a blocker on a website:
Type: ${blocker.type}
Description: ${blocker.description}
Evidence: ${blocker.evidence.join(", ")}

Generate a SKILL.md that would help an autonomous agent handle this situation.
The skill should include:
1. Detection criteria (when to activate)
2. Step-by-step resolution
3. Selectors/elements to interact with (use generic patterns if specific ones unknown)

Return JSON: { "name": "...", "description": "...", "domain": "...", "version": "1.0.0", "content": "...", "triggerCondition": "...", "toolUsage": [...] }`;

	// Use the planner's LLM to generate the skill
	const skill = await this.planner.generateSkill(skillPrompt);
	if (!skill) return false;

	// Write and validate
	const path = await this.skillWriter.createSkill(skill);
	const valid = await this.skillWriter.validateSkill(skill.name);

	if (valid) {
		this.state!.createdSkills.push(skill.name);
		this.emit("skillCreated", { skillName: skill.name, triggeredBy: blocker.type });
		return true;
	}

	return false;
}
```
